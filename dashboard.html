/**
 * Format a date as a half-year label, e.g. "2024-H1".
 * @param {Date} date
 * @returns {string}
 */
function labelHalfYear(date) {
  const yr = date.getFullYear()
  const h = date.getMonth() < 6 ? "H1" : "H2"
  return yr + "-" + h
}


/**
 * Extract sorted unique values from a column in an Arquero table.
 * @param {Object} table - Arquero table
 * @param {string} col - Column name
 * @returns {Array}
 */
function uniqueValues(table, col) {
  return [...new Set(table.array(col))].sort()
}

/**
 * Format a number as a percentage string, e.g. 0.456 → "45.6%".
 * @param {number|null} value
 * @param {number} digits - Decimal places (default: 1)
 * @returns {string}
 */
function formatPercent(value, digits = 1) {
  return value == null ? "–" : value.toFixed(digits) + "%"
}

/**
 * Replace undefined/null values with 0 across all columns of an Arquero table.
 * @param {Object} table - Arquero table
 * @returns {Object} Imputed Arquero table
 */
function fillZero(table) {
  const fills = {}
  for (const col of table.columnNames()) {
    fills[col] = () => 0
  }
  return table.impute(fills)
}

/**
 * Group and aggregate an Arquero table. Sums valueCol if provided,
 * otherwise counts rows.
 * @param {Object} table - Arquero table
 * @param {string[]} groupCols - Columns to group by
 * @param {string|null} valueCol - Column to sum, or null to count rows
 * @returns {Object} Aggregated Arquero table with column "n"
 */
function countGroupwise(table, groupCols, valueCol) {
  if (valueCol) {
    return table
      .rename({ [valueCol]: "_val" })
      .groupby(groupCols)
      .rollup({ n: d => op.sum(d._val) })
      .orderby(groupCols)
  }
  return table
    .groupby(groupCols)
    .rollup({ n: d => op.count() })
    .orderby(groupCols)
}
/**
 * Bin dates into periods (half-year, quarter, year, or split by a cutoff date).
 * @param {Date|string} dateCol - Date column (Date objects or ISO strings)
 * @param {string} binSize - "half-year" (default), "year", or "split"
 * @param {Date|string} [splitDate] - Cutoff date for "split" binning
 * @returns {Object} Object with fields "period" and "period_label"
 */
function binDate(dateStr, timeBin = "half-year", splitDate = null) {
  const d = new Date(dateStr)
  const y = d.getUTCFullYear()
  const m = d.getUTCMonth()  // 0-indexed

  if (timeBin === "year") {
    return { period: `${y}-01-01`, period_label: `${y}` }
  } else if (timeBin === "quarter") {
    const q = Math.floor(m / 3) + 1
    const mStart = String((q - 1) * 3 + 1).padStart(2, "0")
    return { period: `${y}-${mStart}-01`, period_label: `${y}-Q${q}` }
  } else if (timeBin === "split") {
    const splitD = new Date(splitDate)
    const label = d < splitD ? "Before" : "After"
    const sortKey = d < splitD ? "0" : "1"
    return { period: sortKey, period_label: label }
  } else { // half-year default
    const h = m < 6 ? 1 : 2
    const mStart = h === 1 ? "01" : "07"
    return { period: `${y}-${mStart}-01`, period_label: `${y}-H${h}` }
  }
}

/**
 * Return a Leaflet style object for a GeoJSON feature, coloured by
 * a count lookup. States get a lighter border than Locations.
 * @param {Object} f - GeoJSON feature
 * @param {Object} lookup - { regionName: count } plain object
 * @param {number} max - Maximum count for scaling
 * @returns {Object} Leaflet path options
 */
function geoStyle(f, lookup, max) {
  return {
    fillColor: fillColor(lookup[f.properties.name] || 0, max),
    fillOpacity: 0.7,
    weight: 1,
    color: f.properties.type === "state" ? "#999" : "#666"
  };
}

/**
 * Bind a tooltip showing region name and count to a Leaflet layer.
 * @param {Object} f - GeoJSON feature
 * @param {Object} layer - Leaflet layer
 * @param {Object} lookup - { regionName: count } plain object
 */
function geoTooltip(f, layer, lookup) {
  const count = lookup[f.properties.name] || 0;
  layer.bindTooltip(`${f.properties.name}: ${count}`);
}

/**
 * Add a continuous gradient legend to a Leaflet map.
 * @param {Object} map - Leaflet map instance
 * @param {Object} opts
 * @param {Function} opts.interpolator - d3 colour interpolator (default: d3.interpolatePurples)
 * @param {number} opts.max - Upper bound label (default: 1)
 * @param {string} opts.title - Legend heading (default: "Count")
 * @param {string} opts.position - Leaflet control position (default: "bottomleft")
 * @returns {Object} Leaflet control (for later removal if needed)
 */
function gradientLegend(map, {
  interpolator = d3.interpolatePurples,
  max = 1,
  title = "Count",
  position = "bottomleft"
} = {}) {
  const stops = [0, 0.25, 0.5, 0.75, 1]
    .map(t => `${interpolator(t === 0 ? 0 : .15 + (t*.85))} ${t * 100}%`)
    .join(", ");
  const ctrl = L.control({position});
  ctrl.onAdd = () => {
    const div = L.DomUtil.create("div");
    div.innerHTML = `
        <div style="background:var(--bs-body-bg, white); color:var(--bs-body-color, black);
              padding:8px; border-radius:4px;
              box-shadow:0 1px 4px rgba(0,0,0,.3); font-size:11px">
        <div style="margin-bottom:4px"><strong>${title}</strong></div>
        <div style="display:flex; align-items:center; gap:4px">
          <span>0</span>
          <div style="width:80px; height:12px; border-radius:2px;
                      background:linear-gradient(to right, ${stops})"></div>
          <span>${max}</span>
        </div>
      </div>`;
    return div;
  };
  ctrl.addTo(map);
  return ctrl;
}

Protecting identity

The developers of this report acknowledge that projecting molecular epidemiological results to the level of the individual can potentially result in devastating outcomes for PLWH due to criminalisation and stigma. We further believe that trust between the community, public health authorities, and clinical prividers is fundamental to this report being mutually beneficial. To this end, the report was designed under the following constraints, which reflect UNAIDS guidelines on proper use of molecular epidemiology for HIV (UNAIDS 2024).

  • All data incorporated in this report is de-identified via cryptographic hashing of initals and DOB information (Langevin et al. 2026).
  • All clustering methods are non-directional. That is, there is no phylogenetics used in this report to implicate directionality between clustered individuals.
  • Diagrams do not explicitly link individuals, nor is the evolutionary distance between individual samples ever shown. Therefore, for any cluster diagram, one cannot reasonably point to individual transmission events. This step is taken in acknowledgment of the fact that while non-directional clustering does not imply person-to-person transmission, it can still be used alongside circumstantial evidence to implicate individuals.

  1. Raw data consist of notifications and sequences with initials and date of birth replaced by a cryptographic hash identifier, following (Langevin et al. 2026).

  2. Validation
    1. Notifications — Relevant notification fields are formatted and retained. The calculated date of onset, derived from CD4 count, is used as the notification date.
    2. Sequences — Sequence records are validated against a minimum length of bases and a minimum of valid A/T/G/C bases.

  3. Select earliest sequence — For each hash identifier, the earliest sample collection date is retained as a proxy for the transmitted virus. Where multiple sequences with the same hash share the same date of collection, they are excluded form linkage.

  4. Link notifications and sequences — Validated notifications and selected sequences are linked using the hash identifier. A notification can link to one or more sequences, but duplicate notification identifiers are not linked. However, since we filter to transmitted sequences in the step before, linkage is effectively one-to-one. Sequences collected before the calculated onset date or more than 730 days afterwards are excluded.

  5. Prepare sequence regions — Linked prrtin and int sequence fields are converted independently to FASTA format. Each region is submitted to the Stanford HIVdb algorithm via sierrapy for subtype assignment and drug-resistance analysis. The subtype of prrtin is taken as the subtype of a sample.

  6. Alignment and HIV-TRACE clustering — prrtin sequences are aligned to the resources/HXB2_pol.fasta reference using cawlign v0.1.16. The current QC step passes the aligned prrtin sequences through unchanged, since these have passed QC from reference laboratories. HIV-TRACE then computes TN93 genetic distances and identifies clusters using the configured distance threshold and minimum pairwise overlap.

Analysis
Drug Resistance

Drug resistance mutations were identified using the Stanford HIVdb algorithm via sierrapy (v0.4.3), querying the HIVdb database (vHIVDB_10.2 (2026-04-26)) with the Sierra web service (v3.5.7 (2026-09-27)) . High-level resistance (level 5) was retained. Mutations were then filtered down to Surveillance Drug Resistance Mutations (SDRMs) using the curated tables for each drug class, accessed on 13/03/2026 (Stanford HIVdb website 2026).

Clustering

Pairwise genetic distances were computed under the TN93 model using HIV-TRACE (>=0.9.2,<0.10) (Tamura and Nei 1993). Sequences were aligned to the HXB2_pol reference and clustered below a distance threfhold of 0.03. Sequences were compared if they had a minimum pairwise overlap of 800 bp, and ambiguities were treated with the resolve option. Drug resistance mutations were stripped using the lewis option.

Potential contaminants (such as clustering with the reference) and spurious transitive edges are also removed using the --curate remove option and --filter remove options.

Autotune The choice of clustering threshold was investigated using the autotune algorithm (Weaver et al. 2024). Pairwise distances were recomputed over thresholds from 0.0 to 0.05, yielding a score that balances the number and size of clusters. Higher scores indicate more even clustering, with values above 1.9 considered preferable. The threshold used for this report is indicated in Figure 5, and if the optimal threshold diverges substantially from this value, then it may need revision.
